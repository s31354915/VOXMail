cmake_minimum_required(VERSION 3.16)
project(voxmail LANGUAGES C)

# The production build includes this directory from the Baresip source tree,
# while the native harness can be configured directly from this directory.
# Do not make the standalone harness depend on Baresip's private headers.
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  set(_voxmail_build_module_default OFF)
else()
  set(_voxmail_build_module_default ON)
endif()
option(VOXMAIL_BUILD_MODULE "Build the production Baresip module" ${_voxmail_build_module_default})

set(SRCS voxmail.c command_queue.c call_dispatch.c json_output.c frame_parser.c
        client_owner.c command_wakeup.c nonblocking_pipe.c pcm_writer.c
        pthread_start.c)
if(VOXMAIL_BUILD_MODULE)
  list(APPEND MODULES_DETECTED ${PROJECT_NAME})
  set(MODULES_DETECTED ${MODULES_DETECTED} PARENT_SCOPE)
  if(STATIC)
    add_library(${PROJECT_NAME} OBJECT ${SRCS})
  else()
    add_library(${PROJECT_NAME} MODULE ${SRCS})
  endif()
  target_compile_definitions(${PROJECT_NAME} PRIVATE _POSIX_C_SOURCE=200809L)
endif()

option(VOXMAIL_BUILD_NATIVE_TESTS "Build standalone native shim tests" OFF)
if(VOXMAIL_BUILD_NATIVE_TESTS)
  find_package(Threads REQUIRED)
  enable_testing()

  function(voxmail_configure_native_test target)
    target_compile_definitions(${target} PRIVATE _POSIX_C_SOURCE=200809L)
    # The production module is built as Release, but these tests use assert()
    # for both verification and setup.  Keep assertions enabled for the test
    # executables so Release builds do not silently skip their test bodies.
    target_compile_options(${target} PRIVATE -UNDEBUG)
  endfunction()

  add_executable(voxmail_command_queue_test command_queue_test.c command_queue.c call_dispatch.c)
  voxmail_configure_native_test(voxmail_command_queue_test)
  target_link_libraries(voxmail_command_queue_test PRIVATE Threads::Threads)
  add_test(NAME voxmail_command_queue_test COMMAND voxmail_command_queue_test)
  add_executable(voxmail_json_output_test json_output_test.c json_output.c)
  voxmail_configure_native_test(voxmail_json_output_test)
  add_test(NAME voxmail_json_output_test COMMAND voxmail_json_output_test)
  add_executable(voxmail_frame_parser_test frame_parser_test.c frame_parser.c)
  voxmail_configure_native_test(voxmail_frame_parser_test)
  add_test(NAME voxmail_frame_parser_test COMMAND voxmail_frame_parser_test)
  add_executable(voxmail_client_owner_test client_owner_test.c client_owner.c)
  voxmail_configure_native_test(voxmail_client_owner_test)
  target_link_libraries(voxmail_client_owner_test PRIVATE Threads::Threads)
  add_test(NAME voxmail_client_owner_test COMMAND voxmail_client_owner_test)
  add_executable(voxmail_command_wakeup_test command_wakeup_test.c command_wakeup.c)
  voxmail_configure_native_test(voxmail_command_wakeup_test)
  add_test(NAME voxmail_command_wakeup_test COMMAND voxmail_command_wakeup_test)
  add_executable(voxmail_nonblocking_pipe_test nonblocking_pipe_test.c nonblocking_pipe.c)
  voxmail_configure_native_test(voxmail_nonblocking_pipe_test)
  add_test(NAME voxmail_nonblocking_pipe_test COMMAND voxmail_nonblocking_pipe_test)
  add_executable(voxmail_pcm_writer_test pcm_writer_test.c pcm_writer.c)
  voxmail_configure_native_test(voxmail_pcm_writer_test)
  add_test(NAME voxmail_pcm_writer_test COMMAND voxmail_pcm_writer_test)
  add_executable(voxmail_pthread_start_test pthread_start_test.c pthread_start.c)
  voxmail_configure_native_test(voxmail_pthread_start_test)
  target_link_libraries(voxmail_pthread_start_test PRIVATE Threads::Threads)
  add_test(NAME voxmail_pthread_start_test COMMAND voxmail_pthread_start_test)
endif()
